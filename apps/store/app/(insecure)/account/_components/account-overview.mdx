import { ArrowRight, ChevronLeft, ChevronRight, RefreshCw, ShoppingBag } from "lucide-react";
import { Alert, AlertDescription, AlertTitle } from "@86d-app/ui/shadcn/alert";
import { Empty, EmptyContent, EmptyDescription, EmptyHeader, EmptyMedia, EmptyTitle } from "@86d-app/ui/shadcn/empty";
import { Text } from "@86d-app/ui/text";
import { View } from "@86d-app/ui/view";
import Link from "next/link";
import { StatusBadge } from "~/components/status-badge";
import { StoreAction, StoreLink } from "~/components/store-action";
import AccountOverviewSkeleton from "./account-overview-skeleton.mdx";

<section aria-labelledby="recent-orders-heading" className="flex min-w-0 flex-col gap-6" data-testid="account-overview">
  <header className="flex flex-wrap items-start justify-between gap-4">
    <View className="flex flex-col gap-2">
      <Text variant="h2" id="recent-orders-heading" className="text-balance font-display font-medium text-2xl tracking-tight">Recent orders</Text>
      <Text variant="p" className="text-pretty text-sm text-muted-foreground">Check the details and status of your purchases.</Text>
    </View>
    {props.state.status === "ready" && props.state.total > 0 && <StoreLink href="/account/orders" variant="outline">View all orders <ArrowRight data-icon="inline-end" aria-hidden="true" /></StoreLink>}
  </header>

  {props.state.status === "loading" ? <AccountOverviewSkeleton /> : props.state.status === "error" ? (
    <Alert variant="destructive" data-testid="account-orders-error">
      <AlertTitle>{props.state.issue.title}</AlertTitle>
      <AlertDescription>
        <Text variant="p">{props.state.issue.description}</Text>
        <View className="mt-3 flex flex-wrap gap-2">
          {props.state.issue.action && <StoreLink href={props.state.issue.action.href} variant="default">{props.state.issue.action.label}</StoreLink>}
          {props.state.issue.canRetry && <StoreAction static variant="outline" onClick={props.onRetry} disabled={props.isRefreshing}><RefreshCw data-icon="inline-start" aria-hidden="true" />{props.isRefreshing ? "Retrying…" : "Try again"}</StoreAction>}
          <StoreLink href="/track" variant="outline">Track an order</StoreLink>
          {props.canGoBack && <StoreAction static variant="ghost" onClick={props.onPrevious}>Previous page</StoreAction>}
        </View>
      </AlertDescription>
    </Alert>
  ) : (
    <>
      {props.state.orders.length === 0 ? (
        <Empty className="rounded-xl border border-border py-14" data-testid="account-orders-empty">
          <EmptyHeader>
            <EmptyMedia variant="icon"><ShoppingBag aria-hidden="true" /></EmptyMedia>
            <EmptyTitle>{props.state.total === 0 ? "Your first order starts here" : "No orders on this page"}</EmptyTitle>
            <EmptyDescription>{props.state.total === 0 ? "Your purchases will appear here, with their details and current status." : "Return to the previous page or open your full order history."}</EmptyDescription>
          </EmptyHeader>
          <EmptyContent>
            {props.state.total === 0 ? <StoreLink href="/products" variant="default">Explore the store <ArrowRight data-icon="inline-end" aria-hidden="true" /></StoreLink> : <StoreLink href="/account/orders" variant="outline">View all orders</StoreLink>}
          </EmptyContent>
        </Empty>
      ) : (
        <View className="flex min-w-0 flex-col overflow-hidden rounded-xl border border-border" aria-busy={props.isRefreshing} data-testid="account-order-list">
          {props.state.orders.map((order) => (
            <Link key={order.id} href={order.href} className="grid min-w-0 grid-cols-[minmax(0,1fr)_auto] items-start gap-4 border-border border-b p-4 outline-none transition-colors duration-150 last:border-b-0 hover:bg-muted focus-visible:bg-muted focus-visible:ring-2 focus-visible:ring-inset focus-visible:ring-ring sm:p-5">
              <View className="flex min-w-0 flex-col gap-3">
                <Text className="truncate font-medium text-sm" title={order.number}>{order.number}</Text>
                <View className="flex flex-wrap items-center gap-x-3 gap-y-2">
                  <StatusBadge status={order.status} className="max-w-full whitespace-normal break-words" />
                  <Text className="text-xs text-muted-foreground"><time dateTime={order.dateTime}>{order.date}</time></Text>
                </View>
              </View>
              <View className="flex max-w-40 items-center gap-3">
                <Text className="min-w-0 break-words text-right font-medium text-sm tabular-nums">{order.total}</Text>
                <ChevronRight size={16} aria-hidden="true" className="shrink-0 text-muted-foreground" />
              </View>
            </Link>
          ))}
        </View>
      )}
      {(props.state.pages > 1 || props.canGoBack) && (
        <nav aria-label="Recent order pages" className="flex flex-wrap items-center justify-between gap-3">
          <Text className="text-xs text-muted-foreground tabular-nums">{props.state.totalLabel}</Text>
          <View className="flex flex-wrap items-center gap-2">
            <StoreAction static variant="outline" size="icon" aria-label="Previous order page" disabled={!props.canGoBack} onClick={props.onPrevious}><ChevronLeft aria-hidden="true" /></StoreAction>
            <Text role="status" className="px-2 text-sm text-muted-foreground tabular-nums">{props.state.pageLabel}</Text>
            <StoreAction static variant="outline" size="icon" aria-label="Next order page" disabled={!props.canGoForward} onClick={props.onNext}><ChevronRight aria-hidden="true" /></StoreAction>
          </View>
        </nav>
      )}
    </>
  )}
</section>
