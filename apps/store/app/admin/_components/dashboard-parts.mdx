import { Badge } from "@86d-app/ui/badge";
import { Card, CardContent, CardHeader, CardTitle } from "@86d-app/ui/card";
import { Empty, EmptyContent, EmptyDescription, EmptyHeader, EmptyMedia, EmptyTitle } from "@86d-app/ui/shadcn/empty";
import { Item, ItemContent, ItemDescription, ItemGroup, ItemTitle } from "@86d-app/ui/shadcn/item";
import { Skeleton } from "@86d-app/ui/shadcn/skeleton";
import { Text } from "@86d-app/ui/text";
import { View } from "@86d-app/ui/view";
import { ArrowUpRight, Check, ShoppingBag, TriangleAlert } from "lucide-react";
import Link from "next/link";
import { StatusBadge } from "~/components/status-badge";
import { StoreAction, StoreLink } from "~/components/store-action";

export function DashboardMetric({ metric }) { return (
  <Card size="sm" className="min-w-0" data-testid="stat-card" aria-busy={metric.state.status === "loading"}>
    <CardHeader>
      <CardTitle className="text-muted-foreground">{metric.label}</CardTitle>
    </CardHeader>
    <CardContent className="gap-2">
      {metric.state.status === "loading" ? (
        <Skeleton className="h-9 w-24 max-w-full" />
      ) : metric.state.status === "ready" ? (
        <Text variant="p" className="truncate font-semibold text-2xl tabular-nums tracking-tight @3xl/dashboard:text-3xl" data-testid="stat-value" title={metric.state.data}>
          {metric.state.data}
        </Text>
      ) : (
        <Text variant="p" className="flex min-h-9 items-center gap-2 text-muted-foreground text-sm">
          <TriangleAlert className="size-4 shrink-0" aria-hidden="true" /> Unavailable
        </Text>
      )}
      <Text variant="p" className="text-pretty text-muted-foreground text-xs leading-relaxed">{metric.description}</Text>
    </CardContent>
  </Card>
); }

export function DashboardOperation({ metric }) { return (
  <Item render={<Link href={metric.href} />} className="min-w-0 items-start" data-testid="stat-card" aria-busy={metric.state.status === "loading"}>
    <ItemContent className="min-w-0 gap-2">
      <ItemTitle className="w-full justify-between gap-2">
        {metric.label}
        <ArrowUpRight className="size-4 shrink-0 text-muted-foreground" aria-hidden="true" />
      </ItemTitle>
      {metric.state.status === "loading" ? (
        <Skeleton className="h-7 w-16" />
      ) : metric.state.status === "ready" ? (
        <Text className="font-semibold text-xl tabular-nums tracking-tight" data-testid="stat-value">{metric.state.data}</Text>
      ) : (
        <Text className="flex h-7 items-center gap-2 text-muted-foreground text-sm"><TriangleAlert className="size-4" aria-hidden="true" /> Unavailable</Text>
      )}
      <ItemDescription className="text-xs">{metric.description}</ItemDescription>
    </ItemContent>
  </Item>
); }

export function DashboardListLoading() { return (
  <View className="flex flex-col gap-5 py-2" aria-label="Loading records" aria-busy="true">
    {["first", "second", "third"].map((id) => (
      <View key={id} className="flex items-center justify-between gap-4">
        <View className="flex flex-1 flex-col gap-2"><Skeleton className="h-4 w-28" /><Skeleton className="h-3 w-36 max-w-full" /></View>
        <Skeleton className="h-5 w-16 shrink-0" />
      </View>
    ))}
  </View>
); }

export function DashboardListError({ issue, onRefresh, isRefreshing }) { return (
  <Empty className="px-2 py-8">
    <EmptyHeader>
      <EmptyMedia variant="icon"><TriangleAlert aria-hidden="true" /></EmptyMedia>
      <EmptyTitle>{issue.title}</EmptyTitle>
      <EmptyDescription>{issue.description}</EmptyDescription>
    </EmptyHeader>
    <EmptyContent>
      {issue.action ? <StoreLink href={issue.action.href}>{issue.action.label}</StoreLink> : <StoreAction variant="outline" onClick={onRefresh} disabled={isRefreshing || !onRefresh}>{isRefreshing ? "Refreshing…" : "Try again"}</StoreAction>}
    </EmptyContent>
  </Empty>
); }

export function DashboardOrders({ state, onRefresh, isRefreshing }) { return (
  state.status === "loading" ? <DashboardListLoading /> :
  state.status === "error" ? <DashboardListError issue={state.issue} onRefresh={onRefresh} isRefreshing={isRefreshing} /> :
  state.data.length === 0 ? (
    <Empty className="px-4 py-10">
      <EmptyHeader>
        <EmptyMedia variant="icon"><ShoppingBag aria-hidden="true" /></EmptyMedia>
        <EmptyTitle>Your first order starts here</EmptyTitle>
        <EmptyDescription>Add a product, then share your storefront. New orders will appear here.</EmptyDescription>
      </EmptyHeader>
      <EmptyContent><StoreLink href="/admin/products/new" variant="default">Add product</StoreLink></EmptyContent>
    </Empty>
  ) : (
    <ItemGroup className="gap-1">
      {state.data.map((order) => (
        <View key={order.id} role="listitem">
        <Item render={<Link href={order.href} />} className="min-w-0 flex-nowrap">
          <ItemContent className="min-w-0">
            <ItemTitle className="line-clamp-none max-w-full flex-wrap"><Text className="truncate">{order.number}</Text><StatusBadge status={order.status} /></ItemTitle>
            <ItemDescription className="truncate">{order.customer}</ItemDescription>
          </ItemContent>
          <ItemContent className="shrink-0 items-end gap-1 text-right">
            <ItemTitle className="tabular-nums">{order.total}</ItemTitle>
            <ItemDescription className="text-xs tabular-nums">{order.date}</ItemDescription>
          </ItemContent>
        </Item>
        </View>
      ))}
    </ItemGroup>
  )
); }

export function DashboardStock({ state, onRefresh, isRefreshing }) { return (
  state.status === "loading" ? <DashboardListLoading /> :
  state.status === "error" ? <DashboardListError issue={state.issue} onRefresh={onRefresh} isRefreshing={isRefreshing} /> :
  state.data.items.length === 0 ? (
    <Empty className="px-2 py-10">
      <EmptyHeader>
        <EmptyMedia variant="icon"><Check aria-hidden="true" /></EmptyMedia>
        <EmptyTitle>No low-stock alerts</EmptyTitle>
        <EmptyDescription>Items below their stock threshold will appear here.</EmptyDescription>
      </EmptyHeader>
      <EmptyContent><StoreLink href="/admin/inventory" variant="outline">Manage inventory</StoreLink></EmptyContent>
    </Empty>
  ) : (
    <ItemGroup className="gap-1">
      {state.data.items.map((item) => (
        <View key={item.id} role="listitem">
        <Item render={<Link href={item.href} />} className="min-w-0">
          <ItemContent className="min-w-0">
            <ItemTitle className="max-w-full truncate">{item.name}</ItemTitle>
            {item.variant && <ItemDescription className="truncate text-xs">{item.variant}</ItemDescription>}
          </ItemContent>
          <Badge variant={item.outOfStock ? "destructive" : "caution"} className="tabular-nums">{item.available}</Badge>
        </Item>
        </View>
      ))}
      {state.data.remaining && <Text variant="p" className="px-4 pt-2 text-muted-foreground text-xs">{state.data.remaining}</Text>}
    </ItemGroup>
  )
); }
