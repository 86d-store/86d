# Delivery Bookings

View and manage customer delivery slot bookings.

<div className="space-y-4">
  <div className="flex flex-wrap items-center gap-2">
    <input
      type="date"
      value={props.dateFilter}
      onChange={(e) => props.onDateFilterChange(e.target.value)}
      className="rounded-md border border-border bg-background px-3 py-1.5 text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring"
    />
    <select
      value={props.statusFilter}
      onChange={(e) => props.onStatusFilterChange(e.target.value)}
      className="rounded-md border border-border bg-background px-3 py-1.5 text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring"
    >
      <option value="">All statuses</option>
      <option value="pending">Pending</option>
      <option value="confirmed">Confirmed</option>
      <option value="completed">Completed</option>
      <option value="cancelled">Cancelled</option>
    </select>
    {(props.dateFilter || props.statusFilter) && (
      <button
        type="button"
        onClick={() => { props.onDateFilterChange(""); props.onStatusFilterChange(""); }}
        className="text-muted-foreground text-xs hover:text-foreground"
      >
        Clear filters
      </button>
    )}
  </div>

  {props.cancelError && (
    <div className="rounded-md border border-destructive/30 bg-destructive/10 p-3 text-destructive text-sm">{props.cancelError}</div>
  )}

  {props.loading ? (
    <div className="py-12 text-center text-muted-foreground text-sm">Loading bookings…</div>
  ) : props.bookings.length === 0 ? (
    <div className="rounded-lg border border-border bg-card p-10 text-center">
      <p className="font-medium text-foreground text-sm">No bookings found</p>
      <p className="mt-1 text-muted-foreground text-xs">
        {props.dateFilter || props.statusFilter ? "Try adjusting your filters." : "Delivery slot bookings will appear here."}
      </p>
    </div>
  ) : (
    <div className="overflow-x-auto rounded-lg border border-border">
      <table className="w-full text-left text-sm">
        <thead className="border-border border-b bg-muted/50">
          <tr>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Delivery Date</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Order</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Customer</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Status</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Instructions</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Booked</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs" />
          </tr>
        </thead>
        <tbody className="divide-y divide-border">
          {props.bookings.map((b) => (
            <tr key={b.id} className="hover:bg-muted/30">
              <td className="px-4 py-3 font-medium text-foreground">{b.deliveryDate}</td>
              <td className="px-4 py-3 font-mono text-muted-foreground text-xs">{b.orderId.slice(0, 8)}…</td>
              <td className="px-4 py-3 font-mono text-muted-foreground text-xs">{b.customerId.slice(0, 8)}…</td>
              <td className="px-4 py-3">
                <span className={`inline-flex rounded-full px-2 py-0.5 font-medium text-xs ${
                  b.status === "confirmed" ? "bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400" :
                  b.status === "completed" ? "bg-blue-100 text-blue-800 dark:bg-blue-900/30 dark:text-blue-400" :
                  b.status === "cancelled" ? "bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400" :
                  "bg-yellow-100 text-yellow-800 dark:bg-yellow-900/30 dark:text-yellow-400"
                }`}>
                  {b.status}
                </span>
              </td>
              <td className="max-w-xs px-4 py-3 text-muted-foreground text-xs">
                {b.instructions ? (
                  <span className="truncate block">{b.instructions}</span>
                ) : (
                  <span className="text-muted-foreground/50">—</span>
                )}
              </td>
              <td className="px-4 py-3 text-muted-foreground text-xs">
                {new Date(b.createdAt).toLocaleDateString()}
              </td>
              <td className="px-4 py-3">
                {b.status !== "cancelled" && b.status !== "completed" && (
                  <button
                    type="button"
                    disabled={props.cancellingId === b.id}
                    onClick={() => props.onCancel(b.id)}
                    className="text-muted-foreground text-xs hover:text-destructive disabled:opacity-50"
                  >
                    {props.cancellingId === b.id ? "Cancelling…" : "Cancel"}
                  </button>
                )}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )}

  <div className="flex items-center justify-between">
    <button
      type="button"
      disabled={props.skip === 0}
      onClick={props.onPrev}
      className="rounded-md border border-border px-3 py-1.5 text-foreground text-sm hover:bg-muted disabled:opacity-50"
    >
      Previous
    </button>
    <span className="text-muted-foreground text-xs">
      {props.skip + 1}–{props.skip + props.bookings.length}
    </span>
    <button
      type="button"
      disabled={!props.hasMore}
      onClick={props.onNext}
      className="rounded-md border border-border px-3 py-1.5 text-foreground text-sm hover:bg-muted disabled:opacity-50"
    >
      Next
    </button>
  </div>
</div>
