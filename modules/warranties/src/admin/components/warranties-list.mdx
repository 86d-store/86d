# Warranty Claims

Manage warranty claims from customers.

<div className="space-y-4">
  {props.summary && (
    <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
      <div className="rounded-lg border border-border bg-card px-4 py-3">
        <p className="text-muted-foreground text-xs">Total Claims</p>
        <p className="font-semibold text-foreground text-2xl tabular-nums">{props.summary.totalClaims}</p>
      </div>
      <div className="rounded-lg border border-border bg-card px-4 py-3">
        <p className="text-muted-foreground text-xs">Pending Review</p>
        <p className="font-semibold text-foreground text-2xl tabular-nums">{props.summary.submitted + props.summary.underReview}</p>
      </div>
      <div className="rounded-lg border border-border bg-card px-4 py-3">
        <p className="text-muted-foreground text-xs">In Progress</p>
        <p className="font-semibold text-foreground text-2xl tabular-nums">{props.summary.approved + props.summary.inRepair}</p>
      </div>
      <div className="rounded-lg border border-border bg-card px-4 py-3">
        <p className="text-muted-foreground text-xs">Resolved</p>
        <p className="font-semibold text-foreground text-2xl tabular-nums">{props.summary.resolved + props.summary.closed}</p>
      </div>
    </div>
  )}

  <div className="flex items-center gap-2">
    <select
      value={props.statusFilter}
      onChange={(e) => props.onStatusChange(e.target.value)}
      className="rounded-md border border-border bg-background px-3 py-1.5 text-foreground text-sm focus:outline-none focus:ring-2 focus:ring-ring"
    >
      <option value="">All statuses</option>
      <option value="submitted">Submitted</option>
      <option value="under_review">Under Review</option>
      <option value="approved">Approved</option>
      <option value="denied">Denied</option>
      <option value="in_repair">In Repair</option>
      <option value="resolved">Resolved</option>
      <option value="closed">Closed</option>
    </select>
  </div>

  {props.loading ? (
    <div className="py-12 text-center text-muted-foreground text-sm">Loading claims…</div>
  ) : props.claims.length === 0 ? (
    <div className="rounded-lg border border-border bg-card p-10 text-center">
      <p className="font-medium text-foreground text-sm">No warranty claims found</p>
      <p className="mt-1 text-muted-foreground text-xs">
        {props.statusFilter ? "Try clearing the status filter." : "Claims submitted by customers will appear here."}
      </p>
    </div>
  ) : (
    <div className="overflow-x-auto rounded-lg border border-border">
      <table className="w-full text-left text-sm">
        <thead className="border-border border-b bg-muted/50">
          <tr>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Customer</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Issue Type</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Status</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs">Submitted</th>
            <th className="px-4 py-2.5 font-medium text-muted-foreground text-xs" />
          </tr>
        </thead>
        <tbody className="divide-y divide-border">
          {props.claims.map((claim) => (
            <tr key={claim.id} className="hover:bg-muted/30">
              <td className="px-4 py-3 text-foreground text-sm">{claim.customerEmail ?? claim.customerId.slice(0, 8) + "…"}</td>
              <td className="px-4 py-3 text-foreground text-sm capitalize">{claim.issueType.replace(/_/g, " ")}</td>
              <td className="px-4 py-3">
                <span className={`inline-flex rounded-full px-2 py-0.5 font-medium text-xs ${props.statusColors[claim.status] ?? "bg-muted text-muted-foreground"}`}>
                  {claim.status.replace(/_/g, " ")}
                </span>
              </td>
              <td className="px-4 py-3 text-muted-foreground text-xs">
                {new Date(claim.submittedAt).toLocaleDateString()}
              </td>
              <td className="px-4 py-3">
                <a
                  href={`/admin/warranties/claims/${claim.id}`}
                  className="text-muted-foreground text-xs hover:text-foreground"
                >
                  View
                </a>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )}
</div>
