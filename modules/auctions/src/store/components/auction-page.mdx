import Link from "next/link";

{props.loading && (
  <div className="py-12 text-center text-muted-foreground text-sm">Loading auction…</div>
)}

{!props.loading && !props.auction && (
  <div className="flex flex-col items-center justify-center py-24 text-center">
    <p className="font-medium text-foreground text-sm">Auction not found</p>
    <Link href="/auctions" className="mt-2 text-muted-foreground text-sm transition-colors hover:text-foreground">
      Browse auctions
    </Link>
  </div>
)}

{!props.loading && props.auction && (
  <div className="py-4 sm:py-6">
    <div className="grid gap-8 lg:grid-cols-2 lg:gap-12">
      {/* Image */}
      <div className="aspect-square overflow-hidden rounded-lg bg-muted">
        {props.auction.imageUrl ? (
          <img
            src={props.auction.imageUrl}
            alt={props.auction.title}
            className="h-full w-full object-cover"
          />
        ) : (
          <div className="flex h-full items-center justify-center text-muted-foreground/30">
            <svg xmlns="http://www.w3.org/2000/svg" width="48" height="48" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
              <rect width="18" height="18" x="3" y="3" rx="2" />
              <circle cx="9" cy="9" r="2" />
              <path d="m21 15-3.086-3.086a2 2 0 0 0-2.828 0L6 21" />
            </svg>
          </div>
        )}
      </div>

      {/* Info + bidding */}
      <div className="flex flex-col gap-5">
        <div>
          <p className="text-muted-foreground text-xs uppercase tracking-wide">{props.auction.productName}</p>
          <h1 className="mt-1 font-display font-semibold text-foreground text-xl tracking-tight sm:text-2xl">
            {props.auction.title}
          </h1>
          {props.auction.description && (
            <p className="mt-2 text-muted-foreground text-sm leading-relaxed">{props.auction.description}</p>
          )}
        </div>

        {/* Status + timer */}
        <div className="flex flex-wrap items-center gap-3">
          <span className={`rounded-full px-2.5 py-0.5 font-medium text-xs ${props.auction.status === "active" ? "bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400" : props.auction.status === "ended" ? "bg-muted text-muted-foreground" : "bg-yellow-100 text-yellow-800 dark:bg-yellow-900/30 dark:text-yellow-400"}`}>
            {props.auction.status}
          </span>
          <span className="text-muted-foreground text-xs">
            Ends {new Date(props.auction.endsAt).toLocaleString()}
          </span>
        </div>

        {/* Bid summary */}
        <div className="grid grid-cols-2 gap-3">
          <div className="rounded-lg border border-border bg-card p-4">
            <p className="text-muted-foreground text-xs">Current bid</p>
            <p className="mt-1 font-display font-bold text-foreground text-2xl tabular-nums">
              ${(props.auction.currentBid / 100).toFixed(2)}
            </p>
            <p className="mt-0.5 text-muted-foreground text-xs">{props.auction.bidCount} bid{props.auction.bidCount !== 1 ? "s" : ""}</p>
          </div>
          {props.auction.buyNowPrice > 0 && (
            <div className="rounded-lg border border-border bg-card p-4">
              <p className="text-muted-foreground text-xs">Buy it now</p>
              <p className="mt-1 font-display font-bold text-foreground text-2xl tabular-nums">
                ${(props.auction.buyNowPrice / 100).toFixed(2)}
              </p>
            </div>
          )}
        </div>

        {/* Bid form */}
        {props.isActive && (
          <div className="flex flex-col gap-3">
            {props.bidSuccess && (
              <div className="rounded-md bg-green-50 px-3 py-2 text-green-800 text-sm dark:bg-green-900/20 dark:text-green-300">
                {props.bidSuccess}
              </div>
            )}
            {props.bidError && (
              <div className="rounded-md border border-destructive/30 bg-destructive/5 px-3 py-2 text-destructive text-sm">
                {props.bidError}
              </div>
            )}
            <form onSubmit={props.onPlaceBid} className="flex flex-col gap-3">
              <div>
                <label htmlFor="bid-amount" className="mb-1.5 block text-foreground text-sm">
                  Your bid <span className="text-muted-foreground">(min ${props.minBidDollars})</span>
                </label>
                <div className="flex items-center gap-2">
                  <span className="text-muted-foreground text-sm">$</span>
                  <input
                    id="bid-amount"
                    type="number"
                    min={props.minBidDollars}
                    step="0.01"
                    value={props.bidAmount}
                    onChange={(e) => props.onBidAmountChange(e.target.value)}
                    placeholder={props.minBidDollars}
                    required
                    className="w-full rounded-md border border-border bg-background px-3 py-2 text-foreground text-sm focus:border-foreground focus:outline-none"
                  />
                </div>
              </div>
              <div>
                <label htmlFor="max-auto-bid" className="mb-1.5 block text-muted-foreground text-xs">
                  Max auto-bid (optional) — we'll bid on your behalf up to this amount
                </label>
                <div className="flex items-center gap-2">
                  <span className="text-muted-foreground text-sm">$</span>
                  <input
                    id="max-auto-bid"
                    type="number"
                    min={props.minBidDollars}
                    step="0.01"
                    value={props.maxAutoBid}
                    onChange={(e) => props.onMaxAutoBidChange(e.target.value)}
                    placeholder="Optional"
                    className="w-full rounded-md border border-border bg-background px-3 py-2 text-foreground text-sm focus:border-foreground focus:outline-none"
                  />
                </div>
              </div>
              <button
                type="submit"
                disabled={props.isPending}
                className="rounded-md bg-foreground py-2.5 font-medium text-background text-sm transition-opacity hover:opacity-85 disabled:opacity-40"
              >
                {props.isPending ? "Placing bid…" : "Place bid"}
              </button>
            </form>
            {props.auction.buyNowPrice > 0 && (
              <button
                type="button"
                onClick={props.onBuyNow}
                disabled={props.isPending}
                className="rounded-md border border-border py-2.5 font-medium text-foreground text-sm transition-colors hover:bg-muted disabled:opacity-40"
              >
                Buy now — ${(props.auction.buyNowPrice / 100).toFixed(2)}
              </button>
            )}
          </div>
        )}

        {!props.isActive && props.auction.status === "ended" && (
          <div className="rounded-md border border-border bg-muted/30 px-4 py-3">
            <p className="font-medium text-foreground text-sm">This auction has ended</p>
            {props.bids.length > 0 && props.bids[0].isWinning && (
              <p className="mt-1 text-muted-foreground text-sm">
                Winning bid: ${(props.bids[0].amount / 100).toFixed(2)}
                {props.bids[0].customerName ? ` by ${props.bids[0].customerName}` : ""}
              </p>
            )}
          </div>
        )}
      </div>
    </div>

    {/* Bid history */}
    {props.bids.length > 0 && (
      <div className="mt-10">
        <h2 className="mb-4 font-semibold text-foreground text-sm">Bid history</h2>
        <div className="overflow-hidden rounded-lg border border-border">
          <table className="w-full text-sm">
            <thead className="border-border border-b bg-muted/50">
              <tr>
                <th className="px-4 py-2.5 text-left font-medium text-muted-foreground text-xs">Bidder</th>
                <th className="px-4 py-2.5 text-right font-medium text-muted-foreground text-xs">Amount</th>
                <th className="hidden px-4 py-2.5 text-right font-medium text-muted-foreground text-xs sm:table-cell">Time</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-border">
              {props.bids.map((bid) => (
                <tr key={bid.id} className={bid.isWinning ? "bg-green-50/50 dark:bg-green-900/10" : ""}>
                  <td className="px-4 py-3 text-foreground">
                    <span className="flex items-center gap-1.5">
                      {bid.isWinning && <span className="text-green-600 text-xs">★</span>}
                      {bid.customerName ?? "Anonymous"}
                    </span>
                  </td>
                  <td className="px-4 py-3 text-right font-medium text-foreground tabular-nums">
                    ${(bid.amount / 100).toFixed(2)}
                  </td>
                  <td className="hidden px-4 py-3 text-right text-muted-foreground text-xs sm:table-cell">
                    {new Date(bid.createdAt).toLocaleString()}
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    )}
  </div>
)}
