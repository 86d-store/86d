import Link from "next/link";

<div className="py-4 sm:py-6">
  <div className="mb-6">
    <h1 className="font-display font-semibold text-foreground text-2xl tracking-tight">Auctions</h1>
    <p className="mt-1 text-muted-foreground text-sm">Browse active auctions and place your bids.</p>
  </div>

  {props.loading && (
    <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
      {(["k0", "k1", "k2", "k3", "k4", "k5"]).map((key) => (
        <div key={key} className="animate-pulse overflow-hidden rounded-lg border border-border bg-card">
          <div className="aspect-[4/3] bg-muted" />
          <div className="space-y-2 p-4">
            <div className="h-3 w-2/3 rounded bg-muted" />
            <div className="h-5 w-1/2 rounded bg-muted" />
            <div className="h-3 w-1/3 rounded bg-muted" />
          </div>
        </div>
      ))}
    </div>
  )}

  {!props.loading && props.auctions.length === 0 && (
    <div className="flex flex-col items-center justify-center rounded-lg border border-border bg-muted/30 py-24 text-center">
      <svg xmlns="http://www.w3.org/2000/svg" width="32" height="32" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" className="mb-3 text-muted-foreground/40" aria-hidden="true">
        <path d="M12 2v20M17 5H9.5a3.5 3.5 0 0 0 0 7h5a3.5 3.5 0 0 1 0 7H6" />
      </svg>
      <p className="font-medium text-foreground text-sm">No active auctions</p>
      <p className="mt-1 text-muted-foreground text-sm">Check back soon for new listings.</p>
    </div>
  )}

  {!props.loading && props.auctions.length > 0 && (
    <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
      {props.auctions.map((auction) => {
        const endsAt = new Date(auction.endsAt);
        const now = new Date();
        const diffMs = endsAt.getTime() - now.getTime();
        const diffH = Math.floor(diffMs / 3600000);
        const diffM = Math.floor((diffMs % 3600000) / 60000);
        const timeLeft = diffMs <= 0
          ? "Ended"
          : diffH > 24
            ? `${Math.floor(diffH / 24)}d ${diffH % 24}h`
            : diffH > 0
              ? `${diffH}h ${diffM}m`
              : `${diffM}m`;
        return (
          <Link
            key={auction.id}
            href={`/auctions/${auction.id}`}
            className="group flex flex-col overflow-hidden rounded-lg border border-border bg-card transition-shadow hover:shadow-md"
          >
            <div className="relative aspect-[4/3] overflow-hidden bg-muted">
              {auction.imageUrl ? (
                <img
                  src={auction.imageUrl}
                  alt={auction.title}
                  className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-105"
                />
              ) : (
                <div className="flex h-full items-center justify-center text-muted-foreground/30">
                  <svg xmlns="http://www.w3.org/2000/svg" width="32" height="32" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
                    <rect width="18" height="18" x="3" y="3" rx="2" />
                    <circle cx="9" cy="9" r="2" />
                    <path d="m21 15-3.086-3.086a2 2 0 0 0-2.828 0L6 21" />
                  </svg>
                </div>
              )}
              <div className="absolute top-2 right-2 rounded-full bg-background/80 px-2 py-0.5 font-medium text-foreground text-xs backdrop-blur-sm">
                {timeLeft}
              </div>
            </div>
            <div className="flex flex-1 flex-col gap-1.5 p-4">
              <p className="text-muted-foreground text-xs">{auction.productName}</p>
              <p className="line-clamp-2 font-medium text-foreground text-sm">{auction.title}</p>
              <div className="mt-auto flex items-end justify-between pt-2">
                <div>
                  <p className="text-muted-foreground text-xs">Current bid</p>
                  <p className="font-semibold text-foreground tabular-nums">
                    ${(auction.currentBid / 100).toFixed(2)}
                  </p>
                </div>
                <p className="text-muted-foreground text-xs">{auction.bidCount} bid{auction.bidCount !== 1 ? "s" : ""}</p>
              </div>
              {auction.buyNowPrice > 0 && (
                <p className="text-muted-foreground text-xs">
                  Buy now: <span className="text-foreground">${(auction.buyNowPrice / 100).toFixed(2)}</span>
                </p>
              )}
            </div>
          </Link>
        );
      })}
    </div>
  )}
</div>
